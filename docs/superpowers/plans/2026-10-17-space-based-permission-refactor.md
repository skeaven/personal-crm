# 空间化权限模型重构 Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 将现有的"家庭共享"权限模型重构为"共享空间+用户空间"的双层隔离模型，实现联系人去重、关系私有化和操作审计。

**Architecture:** 引入 `spaces` 表作为数据归属的抽象层，所有业务数据通过 `space_id` 关联到空间。权限判定基于空间类型（shared/user）和用户身份。审计日志通过装饰器自动记录所有修改操作。

**Tech Stack:** 
- Backend: Python 3.11+, FastAPI, SQLAlchemy 2.0, PostgreSQL, Alembic
- Frontend: Vue 3, TypeScript, Element Plus, Pinia
- Testing: pytest, pytest-asyncio

**Spec:** `docs/superpowers/specs/2026-10-17-space-based-permission-refactor-design.md`

## Global Constraints

- Python 版本 >= 3.11
- SQLAlchemy 版本 >= 2.0
- 所有数据库迁移必须包含回滚逻辑
- 所有 API 修改必须保持向后兼容至少 1 个月
- 审计日志的 JSON 字段必须可序列化和可查询
- 联系人名字去重约束：`UNIQUE (family_id, name)`
- 关系必须强制属于用户空间

## Review Focus

1. **联系人名字包含特殊字符**（空格、换行、emoji）：去重约束应基于 trim 后的名字，空名字应拒绝创建
2. **并发创建同名联系人**：两个用户同时创建同名联系人时，数据库唯一约束应捕获冲突并返回友好错误
3. **移动联系人到目标空间已有同名**：应在业务层检查并拒绝，而非等数据库约束触发
4. **审计日志的 JSON 字段包含不可序列化对象**（datetime, Decimal）：应转换为字符串再存储
5. **用户删除后遗留的空间和数据**：应级联删除或转移到家庭共享空间（需明确策略）

from sqlalchemy import BigInteger, CheckConstraint, String, UniqueConstraint
from sqlalchemy.orm import Mapped, mapped_column

class Space(Base, TimestampMixin):
    """空间：数据归属的抽象层，shared（家庭共享）或 user（用户私有）。"""
    
    __tablename__ = "spaces"
    __table_args__ = (
        CheckConstraint(
            "(type = 'shared' AND family_id IS NOT NULL AND owner_user_id IS NULL) OR "
            "(type = 'user' AND owner_user_id IS NOT NULL AND family_id IS NULL)",
            name="check_space_type_consistency"
        ),
        UniqueConstraint("family_id", name="uq_shared_space_per_family", postgresql_where="type = 'shared'"),
        UniqueConstraint("owner_user_id", name="uq_user_space_per_user", postgresql_where="type = 'user'"),
    )
    
    id: Mapped[int] = mapped_column(BigInteger, primary_key=True, autoincrement=True)
    type: Mapped[str] = mapped_column(
        String(20), nullable=False, comment="shared（家庭共享）/ user（用户私有）"
    )
    family_id: Mapped[int | None] = mapped_column(

    owner_user_id: Mapped[int | None] = mapped_column(

# backend/scripts/check_duplicate_contacts.py
"""检测并报告重名联系人，用于迁移前评估。"""
import asyncio
from sqlalchemy import select, func
from app.core.db import get_async_engine, AsyncSession
from app.modules.contacts.models import Contact

async def check_duplicates():
    engine = get_async_engine()
    async with AsyncSession(engine) as db:
        stmt = select(
            Contact.family_id,
            Contact.name,
            func.count().label('cnt'),
            func.string_agg(func.cast(Contact.id, sa.String), ', ').label('ids')
        ).group_by(
            Contact.family_id, Contact.name
        ).having(func.count() > 1)
        
        result = await db.execute(stmt)
        duplicates = result.all()
        
        if not duplicates:
            print("✓ 未发现重名联系人")
            return
        
        print(f"⚠ 发现 {len(duplicates)} 组重名联系人：\n")
        for row in duplicates:
            print(f"Family {row.family_id}: '{row.name}' ({row.cnt} 条) - IDs: {row.ids}")
        
        print("\n请手动处理这些重名联系人后再执行迁移。")

if __name__ == "__main__":
    asyncio.run(check_duplicates())
```

# backend/app/services/permission.py（重写整个文件）
"""权限判定模块 V2：基于空间模型。"""
from sqlalchemy import ColumnElement, and_, or_, select
from sqlalchemy.ext.asyncio import AsyncSession

from app.core.errors import NotFoundError, PermissionDeniedError
from app.core.models import SpaceMixin
from app.modules.auth.models import Space, User


async def get_user_accessible_space_ids(db: AsyncSession, user: User) -> list[int]:
    """获取用户可访问的所有 space ID：
    1. 自己的用户空间
    2. 家庭的共享空间
    3. 同家庭其他成员的用户空间（只读）
    """
    # 我的空间（用户空间 + 共享空间）
    stmt = select(Space.id).where(
        or_(
            and_(Space.type == 'user', Space.owner_user_id == user.id),
            and_(Space.type == 'shared', Space.family_id == user.family_id)
        )
    )
    my_spaces = (await db.execute(stmt)).scalars().all()
    
    # 同家庭其他成员的用户空间
    stmt = select(Space.id).where(
        Space.type == 'user',
        Space.owner_user_id.in_(
            select(User.id).where(
                User.family_id == user.family_id,
                User.id != user.id
            )
        )
    )
    other_user_spaces = (await db.execute(stmt)).scalars().all()
    
    return list(my_spaces) + list(other_user_spaces)


def readable_condition(model_cls: type[SpaceMixin], space_ids: list[int]) -> ColumnElement[bool]:
    """构造"当前用户可读"的 SQL 条件"""
    return model_cls.space_id.in_(space_ids)


async def can_write(db: AsyncSession, user: User, record: SpaceMixin) -> bool:
    """判断用户是否可写某条记录"""
    # 加载 space 关联
    stmt = select(Space).where(Space.id == record.space_id)
    space = (await db.execute(stmt)).scalar_one_or_none()
    
    if not space:
        return False
    
    # 1. 自己用户空间的数据：可写
    if space.type == 'user' and space.owner_user_id == user.id:
        return True
    
    # 2. 共享空间的数据：可写
    if space.type == 'shared' and space.family_id == user.family_id:
        return True
    
    # 3. 其他用户空间的数据：不可写
    return False


async def ensure_can_write(db: AsyncSession, user: User, record: SpaceMixin) -> None:
    """写权限校验，不通过抛异常"""
    if not await can_write(db, user, record):
        raise PermissionDeniedError("只能修改自己空间或共享空间的数据")


async def ensure_readable(db: AsyncSession, user: User, record: SpaceMixin) -> None:
    """运行时读校验：不可读一律按 404 处理"""
    accessible_space_ids = await get_user_accessible_space_ids(db, user)
    if record.space_id not in accessible_space_ids:
        raise NotFoundError("资源不存在")
```

import inspect
from functools import wraps
from typing import Any
from datetime import datetime
from decimal import Decimal

from sqlalchemy.ext.asyncio import AsyncSession

from app.modules.auth.models import AuditLog, User


def _serialize_for_json(data: dict[str, Any]) -> dict[str, Any]:
    """将数据转换为可 JSON 序列化的格式"""
    result = {}
    for key, value in data.items():
        if isinstance(value, datetime):
            result[key] = value.isoformat()
        elif isinstance(value, Decimal):
            result[key] = float(value)
        elif isinstance(value, (str, int, float, bool, type(None))):
            result[key] = value
        elif isinstance(value, (list, tuple)):
            result[key] = [_serialize_for_json({'v': item})['v'] if isinstance(item, dict) else item for item in value]
        elif isinstance(value, dict):
            result[key] = _serialize_for_json(value)
        else:
            result[key] = str(value)
    return result


async def create_audit_log(
    db: AsyncSession,
    user_id: int,
    action: str,
    entity_type: str,
    entity_id: int,
    old_data: dict[str, Any] | None,
    new_data: dict[str, Any] | None,
    space_id: int | None = None,
    ip_address: str | None = None,
    user_agent: str | None = None,
) -> AuditLog:
    """创建审计日志记录"""
    log = AuditLog(
        user_id=user_id,
        action=action,
        entity_type=entity_type,
        entity_id=entity_id,
        old_data=_serialize_for_json(old_data) if old_data else None,
        new_data=_serialize_for_json(new_data) if new_data else None,
        space_id=space_id,
        ip_address=ip_address,
        user_agent=user_agent,
    )
    db.add(log)
    await db.flush()
    return log


def audit_log(entity_type: str):
    """装饰器：自动记录操作日志"""
    def decorator(func):
        @wraps(func)
        async def wrapper(*args, **kwargs):
            # 解析参数
            sig = inspect.signature(func)
            bound = sig.bind(*args, **kwargs)
            bound.apply_defaults()
            
            db: AsyncSession = bound.arguments.get('db')
            user: User = bound.arguments.get('user')
            
            if not db or not user:
                # 如果没有 db 或 user 参数，不记录日志
                return await func(*args, **kwargs)
            
            # 判断操作类型
            func_name = func.__name__
            if 'create' in func_name:
                action = 'create'
                # 执行原函数
                result = await func(*args, **kwargs)
                
                # 记录日志
                await create_audit_log(
                    db=db,
                    user_id=user.id,
                    action=action,
                    entity_type=entity_type,
                    entity_id=result.id,
                    old_data=None,
                    new_data=result.__dict__.copy(),
                    space_id=getattr(result, 'space_id', None)
                )
                return result
                
            elif 'update' in func_name:
                action = 'update'
                # 获取 ID 参数
                entity_id = bound.arguments.get('contact_id') or bound.arguments.get('id')
                
                # 加载原始记录（假设有对应的 get 函数）
                # 这里简化处理，实际应该从数据库加载
                old_data = {}  # 实际应该查询数据库
                
                # 执行原函数
                result = await func(*args, **kwargs)
                
                # 记录日志
                await create_audit_log(
                    db=db,
                    user_id=user.id,
                    action=action,
                    entity_type=entity_type,
                    entity_id=entity_id,
                    old_data=old_data,
                    new_data=result.__dict__.copy(),
                    space_id=getattr(result, 'space_id', None)
                )
                return result
                
            elif 'delete' in func_name or 'archive' in func_name:
                action = 'delete'
                entity_id = bound.arguments.get('contact_id') or bound.arguments.get('id')
                
                # 加载原始记录
                # 实际应该在删除前查询
                old_data = {}
                
                # 执行原函数
                result = await func(*args, **kwargs)
                
                # 记录日志
                await create_audit_log(
                    db=db,
                    user_id=user.id,
                    action=action,
                    entity_type=entity_type,
                    entity_id=entity_id,
                    old_data=old_data,
                    new_data=None,
                    space_id=None
                )
                return result
            else:
                # 不支持的操作，直接执行
                return await func(*args, **kwargs)
        
        return wrapper
    return decorator

async def move_contact_to_space(
    db: AsyncSession, user: User, contact_id: int, target_space_id: int
) -> Contact:
    """移动联系人到另一个空间"""
    # 1. 加载联系人
    contact = await db.get(Contact, contact_id)
    if not contact:
        raise NotFoundError("联系人不存在")
    
    # 2. 检查写权限
    await ensure_can_write(db, user, contact)
    
    # 3. 验证目标空间
    target_space = await db.get(Space, target_space_id)
    if not target_space:
        raise ValidationError("目标空间不存在")
    
    # 只能在用户空间和共享空间之间移动
    if target_space.type == 'user' and target_space.owner_user_id != user.id:
        raise PermissionDeniedError("只能移动到自己的空间或共享空间")
    
    # 4. 检查目标空间是否已有同名联系人
    existing = await db.execute(
        select(Contact).where(
            Contact.family_id == user.family_id,
            Contact.name == contact.name,
            Contact.space_id == target_space_id
        )
    )
    if existing.scalar_one_or_none():
        raise ValidationError(f"目标空间已有名为「{contact.name}」的联系人")
    
    # 5. 移动
    old_space_id = contact.space_id
    contact.space_id = target_space_id
    
    # 6. 记录审计日志
    await create_audit_log(
        db=db,
        user_id=user.id,
        action='move',
        entity_type='contact',
        entity_id=contact_id,
        old_data={'space_id': old_space_id},
        new_data={'space_id': target_space_id},
        space_id=target_space_id
    )
    
    await db.commit()
    await db.refresh(contact)
    return contact

async def list_audit_logs(
    entity_type: str | None = None,
    entity_id: int | None = None,
    user_id: int | None = None,
    space_id: int | None = None,
    action: str | None = None,

import { defineStore } from 'pinia';
import { ref, computed } from 'vue';
import { getSpaces, type Space } from '@/api/spaces';

export const useSpacesStore = defineStore('spaces', () => {
  const spaces = ref<Space[]>([]);
  const loading = ref(false);
  
  const myUserSpace = computed(() => 
    spaces.value.find(s => s.type === 'user' && s.permission === 'read_write')
  );
  
  const sharedSpace = computed(() => 
    spaces.value.find(s => s.type === 'shared')
  );
  
  const writableSpaces = computed(() => 
    spaces.value.filter(s => s.permission === 'read_write')
  );
  
  async function fetchSpaces() {
    loading.value = true;
    try {
      spaces.value = await getSpaces();
    } finally {
      loading.value = false;
    }
  }
  
  function getSpaceById(id: number) {
    return spaces.value.find(s => s.id === id);
  }
  
  return {
    spaces,
    loading,
    myUserSpace,
    sharedSpace,
    writableSpaces,
    fetchSpaces,
    getSpaceById
  };
